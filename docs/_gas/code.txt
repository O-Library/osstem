function normDate_(v) {
  if (v === null || v === undefined || v === '') return '';
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  var s = String(v).trim();
  var m = s.match(/(\d{4})\s*[-./년]\s*(\d{1,2})\s*[-./월]\s*(\d{1,2})/);
  return m ? m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2) : '';
}

function maskEmail_(e) { e = String(e || ''); var p = e.split('@'); if (p.length < 2) return ''; return p[0].slice(0, 2) + '***@' + p[1]; }

function cellOut_(key, v) {
  if (v instanceof Date) {
    var hasTime = v.getHours() || v.getMinutes() || v.getSeconds();
    return Utilities.formatDate(v, TZ, hasTime ? "yyyy-MM-dd'T'HH:mm:ss" : 'yyyy-MM-dd');
  }
  if (JSON_FIELDS[key]) { var arr = key === 'extensions' || key === 'candidates' || key === 'choices'; if (v === '' || v == null) return arr ? [] : null; try { return JSON.parse(v); } catch (e) { return arr ? [] : null; } }
  if (key === 'status' && STATUS_EN[v]) return STATUS_EN[v];
  if (key === 'keywords') return String(v || '').split(',').map(function (s) { return s.trim().toLowerCase(); }).filter(String);
  if (key === 'keys') return String(v || '').split(' ').filter(String);
  if (key === 'votes' || key === 'hits' || key === 'maxVotes' || key === 'buyCount') return v === '' ? 0 : Number(v);
  if (key === 'loanCount' || key === 'no') return v === '' ? (key === 'no' ? null : 0) : Number(v);
  if (key === 'to' || key === 'cc') return String(v || '').split(',').map(function (s) { return s.trim(); }).filter(String);
  return v === null || v === undefined ? '' : String(v);
}

function cellIn_(key, v) {
  if (v === null || v === undefined) return '';
  if (JSON_FIELDS[key]) return (Array.isArray(v) && !v.length) || v === '' ? '' : JSON.stringify(v);
  if (key === 'status' && STATUS_KO[v]) return STATUS_KO[v];
  if (key === 'keywords') return (v || []).join(', ');
  if (key === 'keys') return (v || []).join(' ');
  if (key === 'to' || key === 'cc') return (v || []).join(', ');
  if (key === 'html') return String(v).slice(0, 20000);
  return v;
}

function table_(name) {
  if (_tables[name]) return _tables[name];
  var def = TABLES[name];
  var sh = db_().getSheetByName(def.sheet);
  if (!sh) sh = ensureSheet_(db_(), def); // 새 기능의 탭은 처음 쓸 때 자동 생성
  var values = sh.getDataRange().getValues();
  var header = values[0] || [];
  var idx = def.cols.map(function (c) { return header.indexOf(c[1]); });
  var rows = [];
  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    if (row.join('') === '') continue;
    var o = { _r: r + 1 };
    def.cols.forEach(function (c, i) { o[c[0]] = idx[i] >= 0 ? cellOut_(c[0], row[idx[i]]) : cellOut_(c[0], ''); });
    rows.push(o);
  }
  _tables[name] = { def: def, sheet: sh, rows: rows, idx: idx, width: header.length };
  return _tables[name];
}

function rows_(name) { return table_(name).rows; }

function toRow_(t, o) {
  var arr = [];
  for (var i = 0; i < t.width; i++) arr.push('');
  t.def.cols.forEach(function (c, i) { if (t.idx[i] >= 0) arr[t.idx[i]] = cellIn_(c[0], o[c[0]]); });
  return arr;
}

// 행 저장: _r 이 있으면 덮어쓰기, 없으면 추가
function save_(name, objs) {
  objs = [].concat(objs).filter(Boolean);
  if (!objs.length) return;
  var t = table_(name);
  var fresh = objs.filter(function (o) { return !o._r; });
  var old = objs.filter(function (o) { return o._r; });
  if (old.length > 25) {
    // 많이 바뀌면 전체를 한 번에 기록
    var maxR = t.rows.reduce(function (m, o) { return Math.max(m, o._r || 0); }, 1);
    var blank = toRow_(t, {}).map(function () { return ''; });
    var all = [];
    for (var r = 2; r <= maxR; r++) all.push(blank.slice());
    t.rows.forEach(function (o) { if (o._r) all[o._r - 2] = toRow_(t, o); });
    if (all.length) t.sheet.getRange(2, 1, all.length, t.width).setValues(all);
  } else {
    old.forEach(function (o) { t.sheet.getRange(o._r, 1, 1, t.width).setValues([toRow_(t, o)]); });
  }
  if (fresh.length) {
    var start = Math.max(t.sheet.getLastRow(), 1) + 1;
    t.sheet.getRange(start, 1, fresh.length, t.width).setValues(fresh.map(function (o) { return toRow_(t, o); }));
    fresh.forEach(function (o, i) { o._r = start + i; if (t.rows.indexOf(o) < 0) t.rows.push(o); });
  }
  bumpVersion_();
}

function bumpVersion_() { try { CacheService.getScriptCache().remove('catalog_v2'); } catch (e) {} }

function withLock_(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try { _tables = {}; return fn(); } finally { SpreadsheetApp.flush(); lock.releaseLock(); }
}

// 큰 탭(메일발송기록)은 필요한 열만 / 마지막 N행만 읽기
function columns_(name, keys) {
  var def = TABLES[name];
  var sh = db_().getSheetByName(def.sheet);
  var last = sh.getLastRow();
  if (last < 2) return [];
  var header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  var cols = keys.map(function (k) { var h = def.cols.filter(function (c) { return c[0] === k; })[0][1]; return header.indexOf(h); });
  var data = cols.map(function (ci) { return ci < 0 ? null : sh.getRange(2, ci + 1, last - 1, 1).getValues(); });
  var out = [];
  for (var r = 0; r < last - 1; r++) { var o = {}; keys.forEach(function (k, i) { o[k] = cellOut_(k, data[i] ? data[i][r][0] : ''); }); out.push(o); }
  return out;
}
function tail_(name, n) {

  var start = Math.max(2, last - n + 1);
  return sh.getRange(start, 1, last - start + 1, width).getValues().map(function (row) {
    var o = {}; def.cols.forEach(function (c) { var i = header.indexOf(c[1]); o[c[0]] = cellOut_(c[0], i >= 0 ? row[i] : ''); }); return o;
  });
}

// KDC(한국십진분류) / ISBN 부가기호 → 오스템 분류
function mapKDC_(kdcRaw) {
  if (!kdcRaw) return null;
  var s = String(kdcRaw).replace(/[^0-9.]/g, '');
  if (!s) return null;
  var k = Math.floor(parseFloat(s));
  function r(leaf, group) { return { leaf: leaf, group: group || (leaf ? groupOf_(leaf) : null) }; }
  if (k < 100) return (k === 4 || k === 5) ? r('1109') : r('401');
  if (k < 200) { if (k >= 180 && k < 190) return r('404'); if (k >= 140 && k < 160) return r('409'); if (k >= 160 && k < 180) return r('403'); return r('402'); }
  if (k < 300) return r(null, '900');
  if (k < 400) {
    if (k === 325) return r('130'); if (k === 327) return r('137'); if (k === 326) return r('123');
    if (k >= 320 && k < 330) return r('125'); if (k >= 330 && k < 340) return r('1005'); if (k === 349) return r('1003');
    if (k >= 340 && k < 350) return r('1006'); if (k >= 360 && k < 370) return r('1007'); if (k >= 370 && k < 380) return r('1500');
    if (k >= 380 && k < 390) return r('407'); return r(null, '1000');
  }
  if (k < 500) { if (k >= 410 && k < 420) return r('803'); if (k >= 440 && k < 450) return r('805'); if (k >= 470) return r('804'); return r('802'); }
  if (k < 600) { if (k === 516 || k === 517) return r('1302'); if (k >= 510 && k < 520) return r('1201'); if (k >= 590) return r('1500'); if (k >= 560 && k < 570) return r('1112'); return r(null, '1300'); }
  if (k < 700) return r('407');
  if (k < 800) { if (k >= 720 && k < 730) return r('502'); if (k >= 730 && k < 740) return r('503'); if (k >= 740 && k < 750) return r('501'); return r('223'); }
  if (k < 900) {
    var lang = Math.floor(k / 10) % 10, form = k % 10;
    if (k === 802) return r('223');
    if (form === 1) return r(lang === 1 ? '703' : null, '700');
    if (form === 3) { if (lang === 1) return r('602'); if (lang === 3) return r('607'); if (lang === 4) return r('604'); if (lang === 5) return r('606'); return r(null, '600'); }
    if (form === 4 || form === 6 || form === 8) { if (lang === 1) return r('701'); if (lang === 3) return r('704'); if (lang === 4) return r('702'); return r(null, '700'); }
    return r(null, '600');
  }
  if (k === 911) return r('307'); if (k === 913) return r('308'); if (k >= 920 && k < 930) return r('302'); if (k === 909) return r('305');
  return r('309');
}

function normStr_(s) { return String(s || '').toLowerCase().replace(/\(.*?\)|\[.*?\]/g, '').replace(/[^0-9a-z가-힣]/g, ''); }
function normTitle_(s) { return normStr_(String(s || '').split(/[:：\-–]/)[0]); }
function bigrams_(s) { var t = normStr_(s), set = {}; for (var i = 0; i < t.length - 1; i++) set[t.slice(i, i + 2)] = 1; if (t.length === 1) set[t] = 1; return set; }
function sim_(a, b) {
  var ka = Object.keys(a), kb = Object.keys(b);
  if (!ka.length || !kb.length) return 0;
  var inter = 0; ka.forEach(function (x) { if (b[x]) inter++; });
  return inter / Math.sqrt(ka.length * kb.length);
}

function findSameBook_(books, input) {

  return books.filter(function (b) { return b.status !== 'discarded' && ((i && b.isbn === i) || (t && normTitle_(b.title) === t)); });
}

function classify_(input) {
  var leafs = leafCategories_();
  var leafSet = {}; leafs.forEach(function (c) { leafSet[c.code] = c; });
  var books = rows_('books').filter(function (b) { return b.status !== 'discarded'; });
  var scores = {}, reasons = {};
  function add(code, pts, why) { if (!leafSet[code]) return; scores[code] = (scores[code] || 0) + pts; (reasons[code] = reasons[code] || []).push(why); }

  var same = findSameBook_(books, input);
  if (same.length) add(same[0].catCode, 1000, '이미 보유 중인 도서 (' + same[0].code + ') — 같은 분류로 추가 권 등록');

  var q = bigrams_(input.title);
  books.map(function (b) { return { b: b, s: sim_(q, bigrams_(b.title)) }; })
    .filter(function (x) { return x.s > 0.25; }).sort(function (x, y) { return y.s - x.s; }).slice(0, 7)
    .forEach(function (x) { add(x.b.catCode, Math.round(x.s * 30), '제목 유사: 「' + x.b.title + '」'); });

  var ak = authorKey_(input.author);
  if (ak && ak.length >= 2) {
    var seen = {};
    books.filter(function (b) { return authorKey_(b.author) === ak; }).forEach(function (b) {
      if (seen[b.catCode]) return; seen[b.catCode] = 1;
      add(b.catCode, 25, '같은 저자의 보유 도서: 「' + b.title + '」');
    });
  }

  var k = mapKDC_(input.kdc);
  if (k) {
    if (k.leaf) add(k.leaf, 30, '국립중앙도서관 분류(KDC ' + input.kdc + ')');
    if (k.group) leafs.forEach(function (c) { if (c.group === k.group && c.code !== k.leaf) add(c.code, 8, 'KDC 대분류 일치'); });
  }

  var title = String(input.title || '').toLowerCase().replace(/\s/g, '');
  var desc = String(input.description || '').toLowerCase().replace(/\s/g, '').slice(0, 1500);
  leafs.forEach(function (c) {
    (c.keywords || []).forEach(function (kw) {
      var kwn = kw.replace(/\s/g, '');
      if (kwn.length < 2 && !/[a-z]/.test(kwn)) return;
      if (title.indexOf(kwn) >= 0) add(c.code, 12, "제목 키워드 '" + kw + "'");
      else if (desc.indexOf(kwn) >= 0) add(c.code, 3, "소개글 키워드 '" + kw + "'");
    });
  });
  var cats = (input.categories || []).join(' ').toLowerCase();
  [[/business|economics|management/, '130'], [/self-help/, '205'], [/psychology/, '404'], [/philosophy/, '402'], [/history/, '309'], [/science/, '802'], [/computers/, '1109'], [/fiction/, '604'], [/medical/, '1201'], [/family|parenting/, '1500']]
    .forEach(function (p) { if (p[0].test(cats)) add(p[1], 10, "Google Books 분류 '" + input.categories.join(', ') + "'"); });

  var ranked = Object.keys(scores).map(function (code) { return [code, scores[code]]; }).sort(function (a, b) { return b[1] - a[1]; }).slice(0, 5)
    .map(function (p) {
      var uniq = []; reasons[p[0]].forEach(function (r) { if (uniq.indexOf(r) < 0) uniq.push(r); });
      return { code: p[0], name: leafSet[p[0]].name, groupName: groupName_(p[0]), score: p[1], reasons: uniq.slice(0, 3) };
    });
  var top = ranked.length ? ranked[0].score : 1;
  ranked.forEach(function (r) { r.confidence = Math.round(r.score / top * 100); });
  if (input.forceCat && !ranked.some(function (r) { return r.code === String(input.forceCat); }) && leafSet[input.forceCat]) {
    ranked.push({ code: String(input.forceCat), name: leafSet[input.forceCat].name, groupName: groupName_(input.forceCat), score: 0, confidence: 0, reasons: ['직접 선택'] });
  }
  ranked.forEach(function (r) { r.preview = nextCode_(r.code, input, Number(input.count) || 1); });
  return { candidates: ranked, sameBooks: same.map(function (b) { return { code: b.code, title: b.title }; }) };
}

// 코드: 소분류-구매번호(3자리)-권번호
function nextCode_(catCode, input, count) {
  var inCat = rows_('books').filter(function (b) { return b.catCode === String(catCode); });
  var same = findSameBook_(inCat, input);
  var purchaseNo, start;
  if (same.length) {
    purchaseNo = same[0].purchaseNo;
    start = inCat.filter(function (b) { return b.purchaseNo === purchaseNo; }).reduce(function (m, b) { return Math.max(m, Number(b.copyNo) || 0); }, 0) + 1;
  } else {
    purchaseNo = ('00' + (inCat.reduce(function (m, b) { return Math.max(m, Number(b.purchaseNo) || 0); }, 0) + 1)).slice(-3);
    start = 1;
  }
  var codes = [];
  for (var i = 0; i < (count || 1); i++) codes.push({ catCode: String(catCode), purchaseNo: purchaseNo, copyNo: String(start + i), code: catCode + '-' + purchaseNo + '-' + (start + i) });
  return { codes: codes, isAdditionalCopy: !!same.length };
}

function stripTags_(s) { return String(s || '').replace(/<[^>]+>/g, '').replace(/&quot;/g, '"').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>').trim(); }
function isbn13_(s) { var parts = String(s || '').split(/\s+/).map(function (x) { return x.replace(/[^0-9Xx]/g, ''); }); return parts.filter(function (p) { return p.length === 13; })[0] || parts.filter(function (p) { return p.length === 10; })[0] || ''; }

// 여러 요청을 병렬로 보내고 JSON 파싱
function fetchJsonAll_(reqs) {
  if (!reqs.length) return [];
  var res;
  try { res = UrlFetchApp.fetchAll(reqs.map(function (r) { return { url: r.url, headers: r.headers || {}, muteHttpExceptions: true }; })); } catch (e) { return reqs.map(function () { return null; }); }
  return res.map(function (x) { try { return x.getResponseCode() === 200 ? JSON.parse(x.getContentText()) : null; } catch (e) { return null; } });
}

function parseKakao_(j) { return ((j && j.documents) || []).map(function (d) { return { source: 'kakao', title: d.title, author: (d.authors || []).join(', '), publisher: d.publisher, isbn: isbn13_(d.isbn), pubDate: String(d.datetime || '').slice(0, 10), cover: d.thumbnail, description: stripTags_(d.contents) }; }); }
function parseNl_(j) { return ((j && j.docs) || []).map(function (d) { return { source: 'nl', title: d.TITLE, author: d.AUTHOR, publisher: d.PUBLISHER, isbn: d.EA_ISBN, pubDate: String(d.PUBLISH_PREDATE || '').replace(/(\d{4})(\d{2})(\d{2})/, '$1-$2-$3'), cover: d.TITLE_URL, description: '', kdc: d.KDC || kdcFromAddCode_(d.EA_ADD_CODE) }; }); }
function parseNaver_(j) { return ((j && j.items) || []).map(function (d) { return { source: 'naver', title: stripTags_(d.title), author: stripTags_(d.author).replace(/\^/g, ', '), publisher: stripTags_(d.publisher), isbn: isbn13_(d.isbn), pubDate: String(d.pubdate || '').replace(/(\d{4})(\d{2})(\d{2})/, '$1-$2-$3'), cover: d.image, description: stripTags_(d.description) }; }); }
function parseGoogle_(j) {

    var v = it.volumeInfo || {};

  });

function bookRequests_(input) {
  var k = apiKeys_();
  var isbn = String(input.isbn || '').replace(/\D/g, '');
  var q = isbn.length >= 10 ? isbn : [input.title, input.author].filter(Boolean).join(' ');
  var reqs = [];
  if (k.kakao) reqs.push({ p: parseKakao_, url: 'https://dapi.kakao.com/v3/search/book?size=10&target=' + (isbn.length >= 10 ? 'isbn' : 'title') + '&query=' + encodeURIComponent(isbn.length >= 10 ? isbn : input.title), headers: { Authorization: 'KakaoAK ' + k.kakao } });
  if (k.nl) reqs.push({ p: parseNl_, url: 'https://www.nl.go.kr/seoji/SearchApi.do?cert_key=' + encodeURIComponent(k.nl) + '&result_style=json&page_no=1&page_size=10&' + (isbn.length >= 10 ? 'isbn=' + isbn : 'title=' + encodeURIComponent(input.title)) });
  if (k.naverId && k.naverSecret) reqs.push({ p: parseNaver_, url: 'https://openapi.naver.com/v1/search/book.json?display=10&query=' + encodeURIComponent(q), headers: { 'X-Naver-Client-Id': k.naverId, 'X-Naver-Client-Secret': k.naverSecret } });
  reqs.push({ p: parseGoogle_, url: 'https://www.googleapis.com/books/v1/volumes?maxResults=10&q=' + encodeURIComponent(isbn.length >= 10 ? 'isbn:' + isbn : 'intitle:' + input.title + (input.author ? ' inauthor:' + String(input.author).split(',')[0] : '')) });
  return reqs;
}

function searchBooks_(input) {
  var reqs = bookRequests_(input);
  var out = fetchJsonAll_(reqs);
  var all = [];
  out.forEach(function (j, i) { all = all.concat(reqs[i].p(j)); });
  // ISBN 기준으로 합치기 (빈 필드는 다른 출처로 채움)
  var map = {}, order = [];
  all.forEach(function (r) {
    var key = r.isbn || r.title;
    if (!map[key]) { map[key] = r; order.push(key); }
    else Object.keys(r).forEach(function (f) { if (!map[key][f] && r[f]) map[key][f] = r[f]; });
  });
  var n = function (s) { return String(s || '').replace(/\s/g, '').toLowerCase(); };
  var score = function (r) { return (input.author && n(r.author).indexOf(n(input.author).slice(0, 3)) >= 0 ? 2 : 0) + (input.publisher && n(r.publisher).indexOf(n(input.publisher)) >= 0 ? 2 : 0) + (n(r.title) === n(input.title) ? 3 : 0) + (r.source === 'kakao' ? 1 : 0); };
  var list = order.map(function (k) { return map[k]; });
  list.forEach(function (r) { if (r.isbn && (!r.cover || r.source === 'google')) r.coverCandidate = kyoboCover_(r.isbn); });
  return list.sort(function (a, b) { return score(b) - score(a); });
}

// 원본 시트에 바코드(ISBN)가 없는 책: 제목·출판사·저자로 찾아서 확실할 때만 ISBN·표지를 채움
//   (카카오·네이버·국립중앙도서관 키가 있을 때 정확도가 높고, 없으면 Google Books만 사용)
function fillByTitle_(limit) {
  var books = rows_('books').filter(function (b) { return !b.isbn && !b.cover && !b.coverCheckedAt && b.status !== 'discarded' && b.title; });
  var byTitle = {}; books.forEach(function (b) { var k = titleKey_(b); (byTitle[k] = byTitle[k] || []).push(b); });
  var keys = Object.keys(byTitle).slice(0, limit || 20);
  var n = function (s) { return String(s || '').toLowerCase().replace(/\(.*?\)|\[.*?\]/g, '').replace(/[^0-9a-z가-힣]/g, ''); };
  var found = 0, changed = [], stamp = today_();
  keys.forEach(function (k) {
    var b0 = byTitle[k][0], hit = null;
    try {
      var main = String(b0.title).split(/[:：]/)[0];
      hit = searchBooks_({ title: main, author: String(b0.author || '').split(/[,\s]/)[0], publisher: b0.publisher }).filter(function (r) {
        var sc = 0;
        if (n(r.title).indexOf(n(main)) === 0) sc += 3;
        if (b0.publisher && n(r.publisher).indexOf(n(b0.publisher).slice(0, 2)) >= 0) sc += 2;
        if (b0.author && n(r.author).indexOf(n(String(b0.author).split(/[,\s]/)[0])) >= 0) sc += 2;
        return r.isbn && String(r.isbn).length === 13 && sc >= 5; // 제목 + (출판사 또는 저자)가 맞아야 채움
      })[0] || null;
    } catch (e) { hit = null; }
    byTitle[k].forEach(function (b) {
      if (hit) { b.isbn = hit.isbn; b.cover = hit.cover && hit.source !== 'google' ? hit.cover : kyoboCover_(hit.isbn); if (!b.description && hit.description) b.description = String(hit.description).slice(0, 3000); }
      b.coverCheckedAt = stamp; changed.push(b);
    });
    if (hit) found++;
  });
  save_('books', changed);
  return { checked: keys.length, found: found, remaining: Object.keys(byTitle).length - keys.length };
}

// 이미지 URL이 실제 이미지인지 확인 (교보 → 카카오/구글 순)
function isImage_(resp) {
  if (!resp || resp.getResponseCode() !== 200) return false;
  var h = resp.getHeaders(); var ct = String(h['Content-Type'] || h['content-type'] || '');
  return /^image\//.test(ct) && resp.getBlob().getBytes().length > 2000;
}

// 표지가 비어 있는 도서를 묶음으로 확인해 표지URL 저장 (매일 자동 + 관리자 버튼)
function fillCovers_(limit) {
  var books = rows_('books').filter(function (b) { return b.isbn && !b.cover && !b.coverCheckedAt && b.status !== 'discarded'; });
  var byIsbn = {}; books.forEach(function (b) { (byIsbn[b.isbn] = byIsbn[b.isbn] || []).push(b); });
  var isbns = Object.keys(byIsbn).slice(0, limit || 120);
  var noIsbn = fillByTitle_(20);
  if (!isbns.length) return { checked: noIsbn.checked, found: noIsbn.found, remaining: noIsbn.remaining, isbnFilled: noIsbn.found };
  var resp;
  try { resp = UrlFetchApp.fetchAll(isbns.map(function (i) { return { url: kyoboCover_(i), muteHttpExceptions: true, followRedirects: true }; })); } catch (e) { resp = []; }
  var found = 0, missing = [];
  isbns.forEach(function (isbn, i) {
    if (isImage_(resp[i])) { byIsbn[isbn].forEach(function (b) { b.cover = kyoboCover_(isbn); }); found++; }
    else missing.push(isbn);
  });
  // 교보에 없으면 카카오(키 있을 때) → Google Books 표지
  if (missing.length) {
    var k = apiKeys_();
    var reqs = missing.map(function (isbn) {
      return k.kakao ? { url: 'https://dapi.kakao.com/v3/search/book?target=isbn&query=' + isbn, headers: { Authorization: 'KakaoAK ' + k.kakao }, p: parseKakao_ }
        : { url: 'https://www.googleapis.com/books/v1/volumes?q=isbn:' + isbn, p: parseGoogle_ };
    });
    var js = fetchJsonAll_(reqs);
    missing.forEach(function (isbn, i) {
      var r = reqs[i].p(js[i])[0];
      if (r && r.cover) { byIsbn[isbn].forEach(function (b) { b.cover = r.cover; if (!b.description && r.description) b.description = String(r.description).slice(0, 3000); }); found++; }
    });
  }
  var stamp = today_();
  var changed = [];
  isbns.forEach(function (isbn) { byIsbn[isbn].forEach(function (b) { b.coverCheckedAt = stamp; changed.push(b); }); });
  save_('books', changed);
  var remaining = Object.keys(byIsbn).length - isbns.length;
  return { checked: isbns.length + noIsbn.checked, found: found + noIsbn.found, remaining: remaining + noIsbn.remaining, isbnFilled: noIsbn.found };
}

function loanView_(l) {
  var t = today_();
  var v = {}; Object.keys(l).forEach(function (k) { if (k !== '_r') v[k] = l[k]; });
  v.dday = l.dueDate ? diffDays_(l.dueDate, t) : null;
  v.overdue = l.status === 'active' && v.dday !== null && v.dday < 0;
  v.overdueDays = v.overdue ? -v.dday : 0;
  v.waitingPickup = l.status === 'active' && !!l.pickupDate && t < l.pickupDate;
  // 소속·이름은 사원명부(현재 기준)를 우선 표시. 대출 당시 소속은 teamAtLoan으로 보존
  var e = l.empNo && empMap_()[normEmp_(l.empNo)];
  if (e) {
    var tn = String(e.team || '').trim(), nn = String(e.name || '').trim();
    if (nn) v.name = nn;
    if (tn && tn !== String(l.team || '').trim()) { v.teamAtLoan = l.team; v.team = tn; }
  }
  v.statusLabel = { active: v.overdue ? '연체 ' + v.overdueDays + '일' : v.waitingPickup ? '수령대기' : '대출중', returning: '반납확인중', returned: '반납완료', cancelled: '취소', lost: '분실' }[l.status] || l.status;
  return v;
}
function openByCode_() { var m = {}; rows_('loans').forEach(function (l) { if (OPEN.indexOf(l.status) >= 0) m[l.bookCode] = l; }); return m; }
function bookState_(b, open) {
  if (b.status === 'lost') return '분실';
  if (b.status === 'discarded') return '폐기';
  var l = open[b.code];
  if (!l) return '대출가능';
  return l.status === 'returning' ? '반납확인중' : '대출중';
}

// ─── 카탈로그 (전체를 한 번에 내려주고 검색은 브라우저에서) ──
function catalog_() {
  var cache = CacheService.getScriptCache();
  var hit = cacheGet_(cache, 'catalog_v2');
  if (hit) return hit;
  var open = openByCode_();
  var waiting = {};
  rows_('reservations').forEach(function (r) { if (r.status === 'waiting') waiting[r.key] = (waiting[r.key] || 0) + 1; });
  var newSince = addDays_(today_(), -settings_().newBookDays);
  var map = {}, order = [];
  rows_('books').forEach(function (b) {
    if (b.status === 'discarded' || !b.code) return;
    var k = titleKey_(b);
    if (!map[k]) {
      map[k] = { key: k, title: b.title, author: b.author, publisher: b.publisher, isbn: b.isbn, cover: b.cover || '', catCode: b.catCode,
        catName: catName_(b.catCode), group: groupOf_(b.catCode), copies: 0, available: 0, loanCount: 0, createdAt: b.createdAt || '', earliestDue: '', codes: [] };
      order.push(k);
    }
    var t = map[k];
    t.codes.push(b.code);
    if (!t.isbn && b.isbn) t.isbn = b.isbn;
    if (b.status === 'lost') return;
    t.copies++;
    t.loanCount += Number(b.loanCount) || 0;
    if (!t.cover && b.cover) t.cover = b.cover;
    if ((b.createdAt || '') > t.createdAt) t.createdAt = b.createdAt;
    var st = bookState_(b, open);
    if (st === '대출가능') t.available++;
    else { var l = open[b.code]; if (l && (!t.earliestDue || l.dueDate < t.earliestDue)) t.earliestDue = l.dueDate; }
  });
  var list = order.map(function (k) { var t = map[k]; t.isNew = !!t.createdAt && t.createdAt.slice(0, 10) >= newSince; t.waiting = waiting[k] || 0; return t; })
    .filter(function (t) { return t.copies > 0; });
  var out = { items: list, groups: categories_().map(function (c) { return { code: c.code, name: c.name, group: c.group, isGroup: c.isGroup, view: c.view }; }) };
  cachePut_(cache, 'catalog_v2', out, 600);
  return out;
}

// 100KB 제한을 넘는 값을 여러 조각으로 캐시
function cachePut_(cache, key, obj, ttl) {
  try {
    var s = JSON.stringify(obj), size = 90000, n = Math.ceil(s.length / size), m = {};
    for (var i = 0; i < n; i++) m[key + '_' + i] = s.slice(i * size, (i + 1) * size);
    m[key] = String(n);
    cache.putAll(m, ttl);
  } catch (e) {}
}
function cacheGet_(cache, key) {
  try {
    var n = Number(cache.get(key)); if (!n) return null;
    var keys = []; for (var i = 0; i < n; i++) keys.push(key + '_' + i);
    var parts = cache.getAll(keys); var s = '';
    for (var j = 0; j < n; j++) { if (parts[keys[j]] == null) return null; s += parts[keys[j]]; }
    return JSON.parse(s);
  } catch (e) { return null; }
}

function titleDetail_(key) {
  var books = rows_('books').filter(function (b) { return titleKey_(b) === key && b.status !== 'discarded'; });
  if (!books.length) throw new AppError('도서를 찾을 수 없습니다.');
  try { enrichOnce_(books); } catch (e) {}
  var open = openByCode_();
  var first = books[0];
  var pick = function (f) { var b = books.filter(function (x) { return x[f]; })[0]; return b ? b[f] : ''; };
  var copies = books.map(function (b) { var l = open[b.code]; return { code: b.code, status: bookState_(b, open), dueDate: l ? l.dueDate : '' }; });
  return {
    key: key, title: first.title, author: first.author, publisher: first.publisher, isbn: pick('isbn'), cover: pick('cover'), description: pick('description'),
    pubDate: pick('pubDate'), catCode: first.catCode, catName: catName_(first.catCode), groupName: groupName_(first.catCode), group: groupOf_(first.catCode),

    loanCount: books.reduce(function (s, b) { return s + (Number(b.loanCount) || 0); }, 0), links: links_(pick('isbn'), first.title),
    together: alsoBorrowed_(key),
  };
}

// 이 책을 빌린 사람들이 함께 많이 빌린 책 (대출 기록만으로 셈, 누가 빌렸는지는 내보내지 않음)
function alsoBorrowed_(key) {
  try {
    var loans = rows_('loans').filter(function (l) { return l.status !== 'cancelled' && l.status !== 'rejected'; });
    var who = function (l) { return l.empNo || (l.name ? l.name + '|' + l.team : ''); };
    var readers = {};
    loans.forEach(function (l) { if (l.titleKey === key && who(l)) readers[who(l)] = 1; });
    var n = {}, people = {};
    loans.forEach(function (l) {
      var w = who(l); if (!w || !readers[w] || l.titleKey === key) return;
      var k2 = w + '#' + l.titleKey; if (people[k2]) return; people[k2] = 1;
      n[l.titleKey] = (n[l.titleKey] || 0) + 1;
    });
    var items = catalog_().items, byKey = {}; items.forEach(function (t) { byKey[t.key] = t; });
    return Object.keys(n).filter(function (k) { return n[k] >= 2 && byKey[k]; })
      .sort(function (a, b) { return n[b] - n[a] || (byKey[b].loanCount - byKey[a].loanCount); }).slice(0, 4)
      .map(function (k) { var t = byKey[k]; return { key: k, title: t.title, author: t.author, isbn: t.isbn, cover: t.cover, catName: t.catName, available: t.available, copies: t.copies, earliestDue: t.earliestDue }; });
  } catch (e) { return []; }
}

function applyLoan_(key, empNo) {
  var who = employee_(empNo);
  var s = settings_();
  var mine = rows_('loans').filter(function (l) { return normEmp_(l.empNo) === who.empNo && OPEN.indexOf(l.status) >= 0; }).map(loanView_);
  if (s.blockWhenOverdue && mine.some(function (l) { return l.overdue; })) throw new AppError('연체 중인 도서가 있어 신규 대출이 제한됩니다. 먼저 반납해 주세요.');
  if (mine.length >= s.maxLoansPerUser) throw new AppError('1인당 최대 ' + s.maxLoansPerUser + '권까지 대출할 수 있습니다.');
  if (mine.some(function (l) { return l.titleKey === key; })) throw new AppError('이미 대출 중인 도서입니다.');
  var open = openByCode_();
  var book = rows_('books').filter(function (b) { return titleKey_(b) === key; })
    .sort(function (a, b) { return Number(a.copyNo) - Number(b.copyNo); })
    .filter(function (b) { return bookState_(b, open) === '대출가능'; })[0];
  if (!book) throw new AppError('지금은 대출 가능한 권이 없습니다. 예약을 이용해 주세요.');
  var queue = rows_('reservations').filter(function (r) { return r.key === key && r.status === 'waiting'; }).sort(function (a, b) { return a.createdAt < b.createdAt ? -1 : 1; });
  if (queue.length && normEmp_(queue[0].empNo) !== who.empNo) throw new AppError('예약 대기자가 있는 도서입니다. 예약을 이용해 주세요.');
  if (queue.length) { queue[0].status = 'fulfilled'; save_('reservations', queue[0]); }
  var loan = createLoan_(book, who, today_(), 'user');
  notify_('applied', [loan]);
  return loanView_(loan);
}

function canExtend_(l) {

function addCategory_(code, name, keywords) {
  code = String(code || '').trim();
  if (!/^\d{3,4}$/.test(code)) throw new AppError('분류번호는 3~4자리 숫자입니다.');
  if (categories_().some(function (c) { return c.code === code; })) throw new AppError('이미 있는 분류번호입니다.');
  save_('categories', { code: code, name: String(name).trim(), keywords: String(keywords || '').split(',').map(function (s) { return s.trim().toLowerCase(); }).filter(String) });
  _cats = null;
  return { ok: true };
}

// ─── 데이터 점검 ─────────────────────────────────────
// 원본 시트에서 반납 처리가 누락된 기록: 같은 권(도서코드)을 이후에 다른 사람이 빌린 기록이 있는데 아직 '대출중'으로 남은 경우
function laterLoans_(l, loans) {
  return loans.filter(function (x) {
    return x !== l && x.bookCode === l.bookCode && x.status !== 'cancelled' &&
      (String(x.loanDate) > String(l.loanDate) || (String(x.loanDate) === String(l.loanDate) && Number(x.no) > Number(l.no)));
  }).sort(function (a, b) { return String(a.loanDate).localeCompare(String(b.loanDate)); });
}
function dataCheck_() {
  var loans = rows_('loans');
  var open = loans.filter(function (l) { return OPEN.indexOf(l.status) >= 0; });
  var stale = [];
  open.forEach(function (l) {
    var later = laterLoans_(l, loans);
    if (later.length) { var v = loanView_(l); v.later = later.slice(0, 3).map(function (x) { return { no: x.no, name: x.name, loanDate: x.loanDate, status: loanView_(x).statusLabel }; }); stale.push(v); }
  });
  var byCode = {};
  open.forEach(function (l) { (byCode[l.bookCode] = byCode[l.bookCode] || []).push(l); });
  var dup = Object.keys(byCode).filter(function (k) { return byCode[k].length > 1; }).map(function (k) { return { bookCode: k, loans: byCode[k].map(loanView_) }; });
  var noEmp = open.filter(function (l) { return !l.empNo; }).map(loanView_);
  return { stale: stale, dup: dup, noEmp: noEmp, openCount: open.length };
}
function fixStale_(ids, admin) {
  var loans = rows_('loans'), done = [];
  [].concat(ids).forEach(function (id) {
    var l = loans.filter(function (x) { return x.id === id; })[0];
    if (!l || OPEN.indexOf(l.status) < 0) return;
    var later = laterLoans_(l, loans);
    if (!later.length) return;
    l.status = 'returned';
    l.returnedAt = later[0].loanDate || today_();
    l.returnedBy = admin || '자동정리';
    l.memo = [l.memo, '[' + today_() + ' 데이터 점검: 이후 ' + later[0].name + '님 대출(' + later[0].loanDate + ') 기록이 있어 반납 처리]'].filter(String).join(' ');
    done.push(l);
  });
  save_('loans', done);
  return { fixed: done.length };
}

}

// 오늘 보낼 리마인더 계산 (이미 보낸 안내는 다시 보내지 않음)
function planReminders_() {
  var s = settings_(), t = today_();
  var sent = {}, lastOver = {};
  columns_('mailLog', ['type', 'status', 'keys']).forEach(function (m) {
    if (m.type !== '반납안내' || (m.status !== '발송')) return;
    (m.keys || []).forEach(function (k) {
      sent[k] = 1;
      var p = k.split('|');
      if (p[1] === 'overdue' && (!lastOver[p[0]] || p[2] > lastOver[p[0]])) lastOver[p[0]] = p[2];
    });
  });
  var th = s.mail.dueSoonDays.map(Number).sort(function (a, b) { return a - b; });
  var byUser = {}, noEmail = [];
  rows_('loans').forEach(function (raw) {
    if (raw.status !== 'active') return;
    var l = loanView_(raw), item = null;
    if (l.overdue) {
      var last = lastOver[l.id];
      if (!last || diffDays_(t, last) >= s.mail.overdueRepeatDays) item = { kind: 'overdue', keys: [l.id + '|overdue|' + t] };
    } else {
      var d = th.filter(function (x) { return l.dday <= x && l.dday >= 0; })[0];
      if (d !== undefined && !sent[l.id + '|due' + d + '|' + l.dueDate]) item = { kind: 'dueSoon', keys: th.filter(function (x) { return x >= d; }).map(function (x) { return l.id + '|due' + x + '|' + l.dueDate; }) };
    }
    if (!item) return;
    var mail = emailOf_(raw);
    if (!mail) { noEmail.push(l); return; }
    var u = byUser[mail] = byUser[mail] || { email: mail, name: l.name, overdue: [], dueSoon: [], keys: [] };
    u[item.kind].push(l); u.keys = u.keys.concat(item.keys);
  });
  return { users: Object.keys(byUser).map(function (k) { return byUser[k]; }), noEmail: noEmail };
}

  } else {
    header.forEach(function (h) { if (cur.indexOf(h) < 0) { sh.getRange(1, sh.getLastColumn() + 1).setValue(h).setFontWeight('bold').setBackground('#ffe8d9'); } });
  }
  sh.getRange(1, 1, sh.getMaxRows(), Math.max(sh.getLastColumn(), header.length)).setNumberFormat('@');
  if (def.sheet === '사원명부' && sh.getLastRow() < 2) {
    sh.getRange('F1').setValue('← 사번·성명·소속·이메일을 붙여넣으세요. 사번으로 대출/조회하고, 알림은 이 이메일로 발송됩니다.').setFontColor('#a8660f');
  }
  return sh;
}

// 사원명부와 대출/예약 기록 연결 (사번이 비어 있는 이관 기록 → 이름·소속으로 찾기)
function matchEmployees_() {
  var emps = rows_('employees');
  var byName = {};
  emps.forEach(function (e) { var n = String(e.name || '').trim(); if (n) (byName[n] = byName[n] || []).push(e); });

  rows_('loans').forEach(function (l) {
    if (l.empNo || !l.name) return;

}

// 기존 시트에서 이관 (같은 도서코드/번호는 갱신)
function migrateFromOldSheet_(overwrite) {
  var old = SpreadsheetApp.openById(OLD_SHEET_ID);
  var res = { categories: 0, books: 0, loans: 0 };
  // 분류
  var catSh = old.getSheetByName('오스템도서분류');
  if (catSh) {
    var have = {}; rows_('categories').forEach(function (c) { have[c.code] = c; });
    var add = [];
    catSh.getRange(1, 1, catSh.getLastRow(), 2).getValues().forEach(function (r) {
      var code = String(r[0]).trim(), name = String(r[1]).trim();
      if (!/^\d{3,4}$/.test(code) || !name) return;
      if (have[code]) { if (have[code].name !== name) { have[code].name = name; add.push(have[code]); } }
      else { var c = { code: code, name: name, keywords: [] }; have[code] = c; add.push(c); res.categories++; }
    });
    save_('categories', add); _cats = null;
  }
  // 도서목록
  var bSh = old.getSheetByName('도서목록');
  if (bSh) {
    var v = bSh.getDataRange().getValues();
    var h = headerIdx_(v, '도서명');
    var g = function (r, k) { return h.map[k] === undefined ? '' : r[h.map[k]]; };
    var existing = {}; rows_('books').forEach(function (b) { existing[b.code] = b; });
    var out = [], seen = {};
    v.slice(h.row + 1).forEach(function (r) {
      var title = String(g(r, '도서명')).trim(), cat = String(g(r, '분류번호')).trim(), pno = String(g(r, '구매번호')).trim(), cno = String(g(r, '도서번호')).trim();
      if (!title || !/^\d+$/.test(cat) || !pno || !cno) return;
      if (/^\d+$/.test(pno) && pno.length < 3) pno = ('00' + pno).slice(-3);
      var code = cat + '-' + pno + '-' + cno;
      var isbn = String(g(r, '바코드')).replace(/\D/g, '');
      if (/^97[89]\d{15}$/.test(isbn)) isbn = isbn.slice(0, 13); // 바코드 뒤에 붙은 부가기호 5자리 제거
      var b = existing[code];
      if (seen[code]) return; // 원본 시트에 같은 도서코드가 두 줄 있으면 처음 줄 기준
      seen[code] = 1;
      if (b && !overwrite) {
        // 이미 있는 책: 바코드가 새로 생겼거나, 시트에서 도서명·저자·출판사를 고쳤으면 따라감
        var ch = false, au = String(g(r, '저자')).trim(), pu = String(g(r, '출판사')).trim();
        if (!b.isbn && isbn) { b.isbn = isbn; ch = true; }
        if (b.title !== title) { b.title = title; ch = true; }
        if (au && b.author !== au) { b.author = au; ch = true; }
        if (pu && b.publisher !== pu) { b.publisher = pu; ch = true; }
        if (ch) { out.push(b); res.updated = (res.updated || 0) + 1; }
        return;
      }
      b = b || { code: code, status: 'active', createdAt: '', cover: '', description: '' };
      b.catCode = cat; b.purchaseNo = pno; b.copyNo = cno; b.title = title; b.author = String(g(r, '저자')).trim(); b.publisher = String(g(r, '출판사')).trim(); b.isbn = isbn;
      b.loanCount = Number(g(r, '대출횟수')) || 0;
      existing[code] = b; out.push(b); res.books++;
    });
    save_('books', out);
  }
  // 대출현황
  var lSh = old.getSheetByName('대출현황');
  if (lSh) {
    var lv = lSh.getDataRange().getValues();
    var lh = headerIdx_(lv, '성명');
    var lg = function (r, k) { return lh.map[k] === undefined ? '' : r[lh.map[k]]; };
    var t = today_();
    // 원본 시트에는 같은 번호가 서로 다른 대출에 쓰인 경우가 있어 번호+도서코드로 찾는다
    var byNo = {}; rows_('loans').forEach(function (l) { if (l.no) byNo[l.no + '|' + l.bookCode] = l; });
    var bookMap = {}; rows_('books').forEach(function (b) { bookMap[b.code] = b; });
    var lout = [], lostBooks = [];
    lv.slice(lh.row + 1).forEach(function (r) {
      var code = String(lg(r, '도서코드')).trim(), name = String(lg(r, '성명')).trim();
      if (!code || !name) return;
      var no = Number(lg(r, '번호')) || null;
      var ret = normDate_(lg(r, '반납확인일')), checker = String(lg(r, '반납확인자')).trim(), memo = String(lg(r, '메모')).trim();
      var tag = checker + ' ' + memo, status;
      if (/분실/.test(tag)) status = 'lost';
      else if (/중복/.test(tag)) status = 'cancelled';
      else if (ret && ret <= t && !/^\*/.test(checker)) status = 'returned';
      else if (ret && ret < t) status = 'returned';
      else status = 'active';
      var loanDate = normDate_(lg(r, '대출일'));
      var b = bookMap[code];
      var l = no && byNo[no + '|' + code];
      if (l && !overwrite && l.status !== 'active') return;
      if (l && !overwrite && status === 'active') return;
      l = l || { id: uid_(), no: no, empNo: '', email: '', extensions: [], extensionRequest: null, createdBy: 'sheet' };
      l.bookCode = code; l.titleKey = b ? titleKey_(b) : code.split('-').slice(0, 2).join('-'); l.bookTitle = String(lg(r, '도서명')).trim() || (b ? b.title : '');
      l.team = String(lg(r, '소속')).trim(); l.name = name; l.loanDate = loanDate; l.pickupDate = loanDate ? addDays_(loanDate, 1) : '';
      l.dueDate = normDate_(lg(r, '반납예정일')); l.status = status; l.appliedAt = loanDate;
      l.returnedAt = status === 'returned' ? ret : ''; l.returnedBy = status === 'returned' ? checker : '';
      l.memo = [memo, /^\*/.test(checker) ? checker : ''].filter(String).join(' ');
      lout.push(l); res.loans++;
      if (status === 'lost' && b && b.status !== 'lost') { b.status = 'lost'; lostBooks.push(b); }
    });
    save_('loans', lout);
    save_('books', lostBooks);
  }
  recountLoans_();
  matchEmployees_();
  return res;
}

// 원본 시트에는 '대출횟수' 칸이 없어서, 대출 기록으로 권마다 대출 횟수를 다시 셈 ('많이 읽은 책'·인기순·추천에 사용)
function recountLoans_() {
  var n = {};
  rows_('loans').forEach(function (l) { if (l.status !== 'cancelled' && l.status !== 'rejected' && l.bookCode) n[l.bookCode] = (n[l.bookCode] || 0) + 1; });
  var changed = rows_('books').filter(function (b) { var c = n[b.code] || 0; if ((Number(b.loanCount) || 0) < c) { b.loanCount = c; return true; } return false; });
  save_('books', changed);
  return changed.length;
}

function headerIdx_(values, must) {
  for (var i = 0; i < values.length; i++) {
    var row = values[i].map(function (c) { return String(c).replace(/\s/g, ''); });
    if (row.indexOf(must) >= 0) { var map = {}; row.forEach(function (c, j) { if (c && map[c] === undefined) map[c] = j; }); return { row: i, map: map }; }
  }
  throw new AppError("'" + must + "' 머리글을 찾지 못했습니다.");
}

// 사원명부 통째로 바꾸기 (관리자 화면에서 엑셀 붙여넣기/CSV): rows = [[사번, 성명, 소속, 이메일], ...]
function importEmployees_(rows) {
  rows = (rows || []).map(function (r) { return [normEmp_(r[0]), String(r[1] || '').trim(), String(r[2] || '').trim(), String(r[3] || '').trim()]; })
    .filter(function (r) { return r[0] && r[1] && !/사번|EMP/i.test(r[0]); });
  if (!rows.length) throw new AppError('사번·성명이 있는 행을 찾지 못했습니다. 사번·성명·소속·이메일 4열을 붙여넣어 주세요.');
  var seen = {}, dup = [];
  rows.forEach(function (r) { if (seen[r[0]]) dup.push(r[0]); seen[r[0]] = 1; });
  if (dup.length) throw new AppError('사번이 중복됩니다: ' + dup.slice(0, 5).join(', '));
  var sh = db_().getSheetByName('사원명부');
  var last = sh.getLastRow();
  if (last > 1) sh.getRange(2, 1, last - 1, 4).setValues(Array.apply(null, Array(last - 1)).map(function () { return ['', '', '', '']; }));
  sh.getRange(2, 1, rows.length, 4).setValues(rows);
  _tables = {}; _empMap = null;
  var m = matchEmployees_();
  return { count: rows.length, noEmail: rows.filter(function (r) { return !r[3]; }).length, matched: m.matched };
}

var JOSA = ['에서는', '으로는', '에게서', '에게', '에서', '으로', '이랑', '하고', '까지', '부터', '처럼', '이나', '이고', '하는', '해요', '은', '는', '이', '가', '을', '를', '에', '의', '도', '와', '과', '로', '랑', '만'];

function stripJosa_(w) {
  w = String(w || '').trim().toLowerCase();
  for (var i = 0; i < JOSA.length; i++) { var e = JOSA[i]; if (w.length > e.length + 1 && w.slice(-e.length) === e) return w.slice(0, -e.length); }
  return w;
}
function terms_(text) {
  var out = [];
  String(text || '').replace(/[^0-9a-zA-Z가-힣\s]/g, ' ').split(/\s+/).forEach(function (raw) {
    var t = stripJosa_(raw);
    var st = t.replace(/(했는데|하는데|이라서|해서|하면|인데|이라)$/, ''); // 무례해서→무례, 발표하면→발표, 신입인데→신입
    if (st !== t && st.length >= 2) t = st;
    if (ENDINGS.test(t)) return;
    if (t.length > 2 && t.slice(-1) === '들') t = t.slice(0, -1);
    if (t.length >= 2 && STOPWORDS.indexOf(t) < 0 && out.indexOf(t) < 0) out.push(t);
  });
  return out.slice(0, 12);
}
function situationsFrom_(ids, text) {
  var picked = [], t = normStr_(text).replace(/인사이트|인사말|인사성/g, ''), fromText = [];
  var has = function (w) { return t.indexOf(normStr_(w)) >= 0; };
  SITUATIONS.forEach(function (s) {
    if ((ids || []).indexOf(s.id) >= 0) picked.push(s);
    else if (t && s.trig.some(has)) fromText.push(s);
  });
  // '상사·동료'만 나오고 보고·말하기·신입 같은 다른 고민이 있으면 관계 고민으로 보지 않음
  //   예) "상사에게 보고하는 게 어려워요" → 보고 / "상사가 너무 싫어요" → 관계
  if (t && !picked.length && !fromText.length) {
    var w = SITUATIONS.filter(function (s) { return s.weak && s.weak.some(has); });
    if (w.length) fromText.push(w[0]); // '사람이 너무 힘들어요' → 관계 하나만
  }
  fromText.forEach(function (s) { if (picked.indexOf(s) < 0) picked.push(s); });
  return picked;
}

  categories_().forEach(function (c) { kwByCat[c.code] = (c.keywords || []).map(normStr_); });
  return cat.items.map(function (t) {
    return { t: t, tw: String(t.title || '').toLowerCase().split(/[^0-9a-z가-힣]+/).filter(String), title: normStr_(t.title), catName: normStr_(t.catName), desc: normStr_(desc[t.key] || '').slice(0, 1500), catKw: kwByCat[t.catCode] || [] };
  });
}

function scoreBook_(d, prof) {
  var s = 0, hits = 0, kwHit = false, why = { title: [], desc: [], cat: null, catFrom: '', popular: false };
  prof.terms.forEach(function (term) {
    var n = normStr_(term); if (n.length < 2) return;
    if (CONTEXT_WORDS.indexOf(n) >= 0) { if (d.title.indexOf(n) >= 0) s += 0.3; return; }
    var mine = prof.userTerms.indexOf(term) >= 0;
    // 직접 적은 두 글자 말은 낱말 첫머리에서만 인정 ('화를'이 '대화를'에 걸리지 않게)
    var inTitle = mine && n.length <= 2 ? d.tw.some(function (w) { return w.indexOf(n) === 0; }) : d.title.indexOf(n) >= 0;
    var isKw = prof.kwTerms.indexOf(term) >= 0;
    var m = mine ? 1.5 : 1, counts = mine || !prof.modKw[n]; // 직접 적은 말은 더 무겁게, 트렌드 같은 꾸밈 고민의 낱말은 근거 수에서 뺌
    if (inTitle) { s += 4 * m; if (counts) hits++; if (isKw) kwHit = true; why.title.push(term); }
    else if (d.catName.indexOf(n) >= 0) { s += 2.5 * m; if (counts) hits++; if (isKw) kwHit = true; why.catFrom = why.catFrom || 'term'; }
    else if (!(mine && n.length <= 2) && d.desc.indexOf(n) >= 0) { s += 1.5 * m; if (counts) hits++; if (isKw) kwHit = true; why.desc.push(term); }
    else if (d.catKw.indexOf(n) >= 0) { s += 1.5; }
  });
  var c = prof.cats[d.t.catCode] || prof.cats[d.t.group];
  if (c) { s += c.w; why.cat = c; }
  // 사람이 고른 책은 목록 순서대로 맨 앞에
  var pick; for (var q = 0; q < prof.curated.length; q++) if (d.title.indexOf(prof.curated[q]) === 0) { pick = q; break; }
  if (pick !== undefined) return { d: d, score: 100 - pick + (d.t.available > 0 ? 0.5 : 0), rel: 99, why: why, curated: true };
  // 업무 고민에 육아·가족 책이나 소설이 끼지 않게 (본인이 그런 말을 쓴 경우는 예외)
  if (!prof.allowFamily && (d.t.group === '1500' || /엄마|아빠|딸에게|딸은|아들|육아|아이를|부모/.test(d.t.title))) return null;
  if (!prof.allowFiction && /^6/.test(String(d.t.group)) && prof.sits.length) return null;
  // 적은 고민을 하나도 못 알아들었으면(예: '배고프다') 직무 분야만으로 아무 책이나 추천하지 않음.
  //   이때는 직접 적은 낱말이 책 제목·소개와 두 개 이상 맞거나, 관심 키워드 칸에 적은 말이 맞을 때만 추천
  if (!prof.sits.length && prof.freeText && !kwHit && hits < 2) return null;
  // 고민이 잡혔는데 그 고민 분야 밖의 책이면, 낱말이 두 개 이상 맞을 때만 추천
  if (prof.sits.length && !(c && c.from === 'sit') && hits < 2) return null;
  var rel = s;
  if (rel <= 0) return null;
  s += Math.min(Number(d.t.loanCount) || 0, 30) * 0.05;
  if (d.t.isNew) s += 0.4;
  var yr = String(d.t.title).match(/20(1\d|2\d)/); if (yr) s += Math.max(0, Number(yr[0]) - 2018) * 0.2; // 『트렌드 코리아』 같은 연도판은 최신판 먼저
  why.popular = (Number(d.t.loanCount) || 0) >= 5;
  return { d: d, score: s, rel: rel, why: why };
}

function profile_(input) {
  var job = JOB_GROUPS.filter(function (j) { return j.id === input.job; })[0] || null;
  var free = String(input.situation || '');
  var sits = situationsFrom_(input.situations, free);
  var userTerms = terms_(input.keywords).concat(terms_(free).filter(function (t) { return terms_(input.keywords).indexOf(t) < 0; }))
    .filter(function (t) { return CONTEXT_WORDS.indexOf(normStr_(t)) < 0; });
  var kwTerms = terms_(input.keywords);
  var terms = userTerms.slice();
  sits.forEach(function (s) { s.kw.forEach(function (k) { if (terms.indexOf(k) < 0) terms.push(k); }); });
  var cats = {};
  if (job) job.cats.forEach(function (code) { cats[code] = { w: 1.5, from: 'job', label: job.name }; });
  var main = sits.filter(function (s) { return !s.modifier; });
  // '트렌드'는 다른 고민과 같이 나오면 그 주제의 최신 흐름이라는 뜻 → 분야는 주제 쪽만 ('인사 이슈' → 인사 분야 + 트렌드 낱말)
  (main.length ? main : sits).forEach(function (s) { s.cats.forEach(function (code) { cats[code] = { w: 3, from: 'sit', label: s.label }; }); });
  if (job && !userTerms.length && !sits.length) job.kw.forEach(function (k) { if (terms.indexOf(k) < 0) terms.push(k); });
  var modKw = {};
  if (main.length) sits.forEach(function (s) { if (s.modifier) s.kw.forEach(function (k) { modKw[normStr_(k)] = 1; }); });
  main.forEach(function (s) { s.kw.forEach(function (k) { delete modKw[normStr_(k)]; }); });
  var curated = [];
  // 고민마다 한 권씩 번갈아 순서를 매김 (보고 + 신입이면 보고 1, 신입 1, 보고 2 …)
  var lists = (main.length ? main : sits).map(function (x) { return (CURATED[x.id] || []).map(function (t) { return normStr_(t); }); });
  for (var k = 0; k < 12; k++) lists.forEach(function (l) { if (l[k] !== undefined && curated.indexOf(l[k]) < 0) curated.push(l[k]); });
  var raw = free + ' ' + String(input.keywords || '');
  return { job: job, sits: sits, userTerms: userTerms, kwTerms: kwTerms, freeText: !!free.trim(), terms: terms, cats: cats, modKw: modKw, curated: curated,
    allowFamily: /엄마|아빠|딸|아들|육아|아이|부모|가족|자녀/.test(raw),
    allowFiction: sits.some(function (x) { return x.id === 'read'; }) || /소설/.test(raw) };
}

function reasonOf_(x, prof) {
  var t = x.d.t, parts = [];
  if (x.why.cat && x.why.cat.from === 'sit') parts.push('‘' + x.why.cat.label + '’ 고민에 맞는 「' + t.catName + '」 분야');
  else if (x.why.cat && x.why.cat.from === 'job') parts.push(x.why.cat.label + ' 직무에서 자주 찾는 「' + t.catName + '」 분야');
  if (x.why.title.length) parts.push('제목에 ' + x.why.title.slice(0, 2).map(function (w) { return '‘' + w + '’'; }).join('·') + ' 포함');
  else if (x.why.desc.length) parts.push('책 소개에 ' + x.why.desc.slice(0, 2).map(function (w) { return '‘' + w + '’'; }).join('·') + ' 언급');
  if (x.why.popular) parts.push('사내에서 ' + t.loanCount + '번 대출된 인기 도서');
  else if (t.isNew) parts.push('최근 들어온 신간');
  if (!parts.length) parts.push('「' + t.catName + '」 분야 도서');
  return parts.slice(0, 3).join(' · ');
}

function bookCard_(x, prof) {
  var t = x.d.t;
  return { key: t.key, title: t.title, author: t.author, publisher: t.publisher, isbn: t.isbn, cover: t.cover, catCode: t.catCode, catName: t.catName,
    available: t.available, copies: t.copies, earliestDue: t.earliestDue, waiting: t.waiting, loanCount: t.loanCount, reason: reasonOf_(x, prof) };
}

function recommend_(input) {
  input = input || {};
  var prof = profile_(input);
  if (!prof.job && !prof.terms.length && !String(input.situation || input.keywords || '').trim()) throw new AppError('직무 그룹이나 고민·관심 키워드를 하나 이상 알려 주세요.');
  // 적은 고민을 하나도 알아듣지 못하면(예: '배고프다') 직무만으로 엉뚱한 책을 내밀지 않고 다시 물어봄
  var freeText = String(input.situation || '').trim();
  if (freeText && !String(input.keywords || '').trim() && !prof.sits.length && !(input.situations || []).length) {
    var mine = prof.userTerms.map(normStr_).filter(function (t) { return t.length >= 2; });
    var anyHit = mine.length && corpus_().some(function (d) { return mine.some(function (t) { return d.tw.some(function (w) { return w.indexOf(t) === 0; }) || d.catName.indexOf(t) >= 0; }); });
    if (!anyHit) {
      logSearch_({ source: 'recommend', job: prof.job ? prof.job.id : '', situations: [], terms: prof.userTerms, hits: 0, topScore: 0 });
      return { cards: [], matched: 0, unknown: true, text: freeText.slice(0, 40), understood: { job: prof.job ? prof.job.name : '', situations: [], keywords: prof.userTerms }, weak: false };
    }
  }
  var ranked = corpus_().map(function (d) { return scoreBook_(d, prof); }).filter(Boolean)
    .filter(function (x) { return x.rel >= 2.5; })
    .sort(function (a, b) { return b.score - a.score; });
  var used = {}, seenSeries = {}, cards = [];
  function seriesKey(t) { return normTitle_(t.title).replace(/[0-9]+(권|부|편)?$/, '').replace(/(상|하|上|下)$/, '').slice(0, 6); }
  function free(y) { return !used[y.d.t.key] && !seenSeries[seriesKey(y.d.t)]; }
  function take(y) { used[y.d.t.key] = 1; seenSeries[seriesKey(y.d.t)] = 1; }
  var skip = Math.max(0, Number(input.page) || 0) * 3; // [다른 책 보기]를 누를 때마다 다음 3권
  for (var i = 0; i < ranked.length && cards.length < 3; i++) {
    var x = ranked[i];
    if (!free(x)) continue;
    if (skip > 0) { skip--; take(x); continue; }
    take(x);
    var card = bookCard_(x, prof);
    if (x.d.t.available <= 0) {
      // 대출 중이면 지금 빌릴 수 있는 대체 도서: 같은 소분류 → 같은 대분류 → 같은 키워드 → 그 다음 관련 도서
      var pool = ranked.filter(function (y) { return free(y) && y.d.t.available > 0; });
      var alt = pool.filter(function (y) { return y.d.t.catCode === x.d.t.catCode; })[0]
        || pool.filter(function (y) { return y.d.t.group === x.d.t.group || y.why.title.some(function (w) { return x.why.title.indexOf(w) >= 0; }); })[0]
        || pool[0];
      if (alt) { take(alt); card.alt = bookCard_(alt, prof); }
    }
    cards.push(card);
  }
  logSearch_({ source: 'recommend', job: prof.job ? prof.job.id : '', situations: prof.sits.map(function (s) { return s.id; }), terms: prof.userTerms, hits: ranked.length, topScore: ranked[0] ? Math.round(ranked[0].rel * 10) / 10 : 0 });
  return {
    cards: cards, matched: ranked.length, page: Number(input.page) || 0, more: ranked.length > (Number(input.page) || 0) * 3 + 6,
    understood: { job: prof.job ? prof.job.name : '', situations: prof.sits.map(function (s) { return s.label; }), keywords: prof.userTerms },
    weak: ranked.length < 3,
    unclear: prof.freeText && !prof.sits.length && !cards.length,
  };
}

//  · 신청·투표는 사번으로 본인 확인(중복 투표 방지·입고 알림용). 화면에는 신청자를 공개하지 않음

var REQ_OPEN = ['접수', '투표중', '투표완료', '구매확정'];
var MAX_OPEN_REQUESTS_PER_PERSON = 5;

function ownedMatch_(title, author) {
  var q = bigrams_(title), nt = normTitle_(title), best = null;
  catalog_().items.forEach(function (t) {
    var s = normTitle_(t.title) === nt ? 1 : sim_(q, bigrams_(t.title));
    if (s >= 0.82 && (!best || s > best.s)) best = { s: s, t: t };
  });
  return best ? { key: best.t.key, title: best.t.title, author: best.t.author, available: best.t.available, copies: best.t.copies } : null;
}
function sameRequest_(title) {
  var nt = normTitle_(title), q = bigrams_(title);
  return rows_('requests').filter(function (r) { return REQ_OPEN.indexOf(r.status) >= 0 && (normTitle_(r.title) === nt || sim_(q, bigrams_(r.title)) >= 0.85); })[0] || null;
}

function currentRound_() {
  var t = today_();
  return rows_('rounds').filter(function (r) { return r.status === '진행' && r.startDate <= t && t <= r.endDate; })
    .sort(function (a, b) { return String(b.createdAt).localeCompare(String(a.createdAt)); })[0] || null;
}
function tally_(roundId) {
  var m = {};
  rows_('ballots').forEach(function (b) { if (b.roundId === roundId) (b.choices || []).forEach(function (id) { m[id] = (m[id] || 0) + 1; }); });
  return m;
}
function voterCount_(roundId) { return rows_('ballots').filter(function (b) { return b.roundId === roundId && (b.choices || []).length; }).length; }

function requestView_(r, admin) {
  var v = { id: r.id, createdAt: String(r.createdAt), title: r.title, author: r.author, publisher: r.publisher, reason: r.reason,
    status: r.status, roundId: r.roundId, votes: r.votes || 0, adminMemo: r.adminMemo, bookCode: r.bookCode };
  if (admin) { var e = r.empNo && empMap_()[normEmp_(r.empNo)]; v.empNo = r.empNo; v.requester = e ? String(e.name).trim() + ' (' + String(e.team).trim() + ')' : r.empNo; }
  return v;
}

function roundView_(r, empNo) {
  if (!r) return null;
  var counts = tally_(r.id);
  var reqs = {}; rows_('requests').forEach(function (x) { reqs[x.id] = x; });
  var v = { id: r.id, title: r.title, startDate: r.startDate, endDate: r.endDate, maxVotes: r.maxVotes || 3, buyCount: r.buyCount || 0, status: r.status,
    daysLeft: r.endDate ? diffDays_(r.endDate, today_()) : null, voters: voterCount_(r.id),
    candidates: (r.candidates || []).filter(function (id) { return reqs[id]; }).map(function (id) { var q = requestView_(reqs[id]); q.count = counts[id] || 0; return q; }) };
  if (r.result) v.result = r.result;
  if (empNo) { var k = normEmp_(empNo); var b = rows_('ballots').filter(function (x) { return x.roundId === r.id && normEmp_(x.empNo) === k; })[0]; v.myChoices = b ? b.choices : []; }
  return v;
}

function wishBoard_() {
  var round = currentRound_();
  var list = rows_('requests').filter(function (r) { return r.status !== '반려' || diffDays_(today_(), String(r.decidedAt || r.createdAt).slice(0, 10)) <= 60; })
    .sort(function (a, b) { return String(b.createdAt).localeCompare(String(a.createdAt)); }).slice(0, 200).map(function (r) { return requestView_(r); });
  var last = rows_('rounds').filter(function (r) { return r.status === '마감'; }).sort(function (a, b) { return String(b.endDate).localeCompare(String(a.endDate)); })[0];
  return { round: roundView_(round), lastRound: last ? roundView_(last) : null, requests: list };
}

function submitRequest_(input, empNo, force) {
  var who = employee_(empNo);
  var title = String((input && input.title) || '').trim();
  if (title.length < 2) throw new AppError('도서명을 입력해 주세요.');
  if (!force) {
    var own = ownedMatch_(title, input.author);
    if (own) return { owned: own };
    var dup = sameRequest_(title);
    if (dup) return { duplicate: requestView_(dup) };
  }
  var mine = rows_('requests').filter(function (r) { return normEmp_(r.empNo) === who.empNo && REQ_OPEN.indexOf(r.status) >= 0; });
  if (mine.length >= MAX_OPEN_REQUESTS_PER_PERSON) throw new AppError('진행 중인 신청이 ' + mine.length + '건 있습니다. 처리된 뒤 다시 신청해 주세요. (1인 최대 ' + MAX_OPEN_REQUESTS_PER_PERSON + '건)');
  var r = { id: uid_(), createdAt: nowIso_(), title: title, author: String(input.author || '').trim(), publisher: String(input.publisher || '').trim(),
    reason: String(input.reason || '').trim().slice(0, 500), empNo: who.empNo, status: '접수', roundId: '', votes: 0, adminMemo: '', bookCode: '', decidedAt: '' };
  save_('requests', r);
  return { created: requestView_(r) };
}

function vote_(roundId, ids, empNo) {
  var who = employee_(empNo);
  var r = rows_('rounds').filter(function (x) { return x.id === roundId; })[0];
  var t = today_();
  if (!r || r.status !== '진행' || t < r.startDate || t > r.endDate) throw new AppError('지금은 투표 기간이 아닙니다.');
  ids = [].concat(ids || []).filter(function (id, i, a) { return a.indexOf(id) === i; });
  var max = Number(r.maxVotes) || 3;
  if (ids.length > max) throw new AppError('최대 ' + max + '권까지 고를 수 있습니다.');
  if (ids.some(function (id) { return (r.candidates || []).indexOf(id) < 0; })) throw new AppError('투표 후보가 아닌 도서가 있습니다.');
  var b = rows_('ballots').filter(function (x) { return x.roundId === roundId && normEmp_(x.empNo) === who.empNo; })[0];
  if (!b) b = { roundId: roundId, empNo: who.empNo };
  b.choices = ids; b.at = nowIso_();
  save_('ballots', b);
  syncVotes_(r);
  return roundView_(r, who.empNo);
}
function syncVotes_(r) {
  var counts = tally_(r.id), changed = [];
  rows_('requests').forEach(function (q) { if ((r.candidates || []).indexOf(q.id) >= 0 && (q.votes || 0) !== (counts[q.id] || 0)) { q.votes = counts[q.id] || 0; changed.push(q); } });
  save_('requests', changed);
}

function adminWish_() {

}
function updateRequest_(id, patch) {
  var r = rows_('requests').filter(function (x) { return x.id === id; })[0];
  if (!r) throw new AppError('신청 정보를 찾을 수 없습니다.');
  if (patch.status && ['접수', '투표중', '투표완료', '구매확정', '구매완료', '반려', '보유중'].indexOf(patch.status) < 0) throw new AppError('알 수 없는 상태입니다.');
  ['status', 'adminMemo', 'title', 'author', 'publisher'].forEach(function (k) { if (patch[k] !== undefined) r[k] = String(patch[k]); });
  if (patch.status) r.decidedAt = nowIso_();
  save_('requests', r);
  return requestView_(r, true);
}
function createRound_(input, admin) {
  var ids = [].concat(input.candidates || []);
  if (ids.length < 1) throw new AppError('투표 후보 도서를 1권 이상 골라 주세요.');
  if (currentRound_()) throw new AppError('이미 진행 중인 투표가 있습니다. 마감한 뒤 새 투표를 만들어 주세요.');
  var reqs = rows_('requests').filter(function (r) { return ids.indexOf(r.id) >= 0; });
  var start = input.startDate || today_(), end = input.endDate || addDays_(start, 7);
  if (end < start) throw new AppError('마감일이 시작일보다 빠릅니다.');
  var r = { id: uid_(), title: String(input.title || '').trim() || (start.slice(0, 7).replace('-', '년 ') + '월 희망도서 투표'), startDate: start, endDate: end,
    maxVotes: Math.max(1, Math.min(10, Number(input.maxVotes) || 1)), buyCount: Number(input.buyCount) || 0, candidates: ids.filter(function (id) { return reqs.some(function (x) { return x.id === id; }); }),
    status: '진행', createdAt: nowIso_(), createdBy: admin || '', result: null };
  save_('rounds', r);
  reqs.forEach(function (x) { x.status = '투표중'; x.roundId = r.id; x.votes = 0; });
  save_('requests', reqs);
  return roundView_(r);
}
function closeRound_(id) {
  var r = rows_('rounds').filter(function (x) { return x.id === id; })[0];
  if (!r) throw new AppError('투표를 찾을 수 없습니다.');
  var counts = tally_(r.id);
  var reqs = rows_('requests').filter(function (x) { return (r.candidates || []).indexOf(x.id) >= 0; });
  reqs.sort(function (a, b) { return (counts[b.id] || 0) - (counts[a.id] || 0) || String(a.createdAt).localeCompare(String(b.createdAt)); });
  // 투표 결과는 구매 심사 참고용: 자동으로 구매확정하지 않고 '투표완료'로 둔다
  reqs.forEach(function (x) {
    x.votes = counts[x.id] || 0;
    if (x.status === '투표중') { x.status = '투표완료'; x.decidedAt = nowIso_(); }
  });
  save_('requests', reqs);
  r.status = '마감';
  r.result = { voters: voterCount_(r.id), ranking: reqs.map(function (x) { return { id: x.id, title: x.title, votes: x.votes, status: x.status }; }), closedAt: nowIso_() };
  save_('rounds', r);
  return roundView_(r);
}
// 투표 마감일이 지난 진행 중 투표는 매일 자동 마감
function autoCloseRounds_() {
  var t = today_(), n = 0;
  rows_('rounds').forEach(function (r) { if (r.status === '진행' && r.endDate && r.endDate < t) { closeRound_(r.id); n++; } });
  return n;
}

function fulfillRequests_(created) {
  if (!created || !created.length) return 0;
  var b = created[0];
  var nt = normTitle_(b.title), q = bigrams_(b.title);
  var hit = rows_('requests').filter(function (r) { return REQ_OPEN.indexOf(r.status) >= 0 && (normTitle_(r.title) === nt || sim_(q, bigrams_(r.title)) >= 0.85); });
  hit.forEach(function (r) {
    r.status = '구매완료'; r.bookCode = b.code; r.decidedAt = nowIso_();
    var e = r.empNo && empMap_()[normEmp_(r.empNo)];
    if (e && e.email) {
      var s = settings_();
      try {
        sendMail_({ type: '희망도서입고', to: String(e.email).trim(), subject: '[' + s.libraryName + '] 신청하신 「' + b.title + '」이(가) 들어왔어요',
          html: layout_('신청하신 책이 들어왔어요', '<p>' + esc_(String(e.name).trim()) + '님이 신청하신 <b>「' + esc_(b.title) + '」</b>(도서코드 ' + esc_(b.code) + ')이 등록되었습니다.</p><p>사이트에서 바로 대출 신청하실 수 있습니다.' + (props_().getProperty('WEBAPP_URL') ? // 운영자 인사이트: 자주 찾지만 보유가 적은 주제 → 구매 우선순위 제안
// ─────────────────────────────────────────────────────
function ownedForTerm_(corpus, term) {
  var n = normStr_(term); if (n.length < 2) return 0;
  return corpus.filter(function (d) { return d.title.indexOf(n) >= 0 || d.catName.indexOf(n) >= 0 || d.desc.indexOf(n) >= 0; }).length;
}

function insights_(days) {
  days = Math.max(7, Math.min(365, Number(days) || 90));
  var since = addDays_(today_(), -days);
  var logs = rows_('searchLog').filter(function (l) { return String(l.at).slice(0, 10) >= since; });
  var rec = logs.filter(function (l) { return l.source === 'recommend'; }), srch = logs.filter(function (l) { return l.source !== 'recommend'; });

  // 주제(키워드)별 집계
  var topics = {};
  logs.forEach(function (l) {
    String(l.terms || '').split(' ').filter(String).forEach(function (t) {
      var k = stripJosa_(t); if (k.length < 2) return;
      var o = topics[k] = topics[k] || { term: k, count: 0, weak: 0, recommend: 0, search: 0 };
      o.count++; if ((Number(l.hits) || 0) <= 2) o.weak++;
      if (l.source === 'recommend') o.recommend++; else o.search++;
    });
  });
  var corpus = corpus_();
  var reqs = rows_('requests');
  var list = Object.keys(topics).map(function (k) { return topics[k]; }).filter(function (o) { return o.count >= 2; });
  list.forEach(function (o) {
    o.owned = ownedForTerm_(corpus, o.term);
    var n = normStr_(o.term);
    var rel = reqs.filter(function (r) { return normStr_(r.title + ' ' + r.reason).indexOf(n) >= 0; });
    o.requests = rel.length;
    o.votes = rel.reduce(function (s, r) { return s + (Number(r.votes) || 0); }, 0);
    // 수요(검색·추천·신청·득표) 대비 보유가 적을수록 높은 점수
    o.priority = Math.round(((o.count + o.weak) + o.requests * 3 + o.votes * 2) / (1 + o.owned) * 10) / 10;
  });
  var gaps = list.filter(function (o) { return o.owned <= 3; }).sort(function (a, b) { return b.priority - a.priority || b.count - a.count; }).slice(0, 20);

  // 직무·고민 분포
  var jobs = {}, sits = {};
  rec.forEach(function (l) { if (l.job) jobs[l.job] = (jobs[l.job] || 0) + 1; String(l.situations || '').split(' ').filter(String).forEach(function (s) { sits[s] = (sits[s] || 0) + 1; }); });
  var jobName = {}; JOB_GROUPS.forEach(function (j) { jobName[j.id] = j.name; });
  var sitName = {}; SITUATIONS.forEach(function (s) { sitName[s.id] = s.label; });

  // 구매 우선순위 제안: 희망도서(득표·신청) + 수요 높은 미보유 주제
  var wish = reqs.filter(function (r) { return REQ_OPEN.indexOf(r.status) >= 0; }).map(function (r) {
    var n = normStr_(r.title + ' ' + r.reason);
    var demand = list.filter(function (o) { return n.indexOf(normStr_(o.term)) >= 0; }).reduce(function (s, o) { return s + o.count; }, 0);
    return { kind: 'request', id: r.id, title: r.title, author: r.author, status: r.status, votes: Number(r.votes) || 0, demand: demand, score: (Number(r.votes) || 0) * 3 + demand + (r.status === '구매확정' ? 100 : 0) };
  }).sort(function (a, b) { return b.score - a.score; });

  return {
    days: days, since: since,
    totals: { recommend: rec.length, search: srch.length, weakRecommend: rec.filter(function (l) { return (Number(l.hits) || 0) < 3; }).length, zeroSearch: srch.filter(function (l) { return !(Number(l.hits) || 0); }).length,
      requests: reqs.filter(function (r) { return String(r.createdAt).slice(0, 10) >= since; }).length },
    gaps: gaps,
    jobs: Object.keys(jobs).map(function (k) { return { name: jobName[k] || k, count: jobs[k] }; }).sort(function (a, b) { return b.count - a.count; }),
    situations: Object.keys(sits).map(function (k) { return { name: sitName[k] || k, count: sits[k] }; }).sort(function (a, b) { return b.count - a.count; }),
    wish: wish.slice(0, 20),
  };
}

// GitHub Pages 등 외부 화면에서 쓰는 JSON API (POST, Content-Type: text/plain → CORS 사전요청 없이 호출)
function doPost(e) {
  var out;
  try {
    var body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    out = rpc(String(body.name || ''), body.args || [], body.token || '');
  } catch (err) {
    out = JSON.stringify({ ok: false, error: '요청 형식이 올바르지 않습니다.' });
  }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

// 화면 → 서버 단일 진입점. 결과는 JSON 문자열로 돌려줌
function rpc(name, args, token) {
  try {
    args = args || [];
    var admin = null;
    if (ADMIN_API[name]) {
      admin = token && CacheService.getScriptCache().get('adm_' + token);
      if (!admin) throw new AppError('관리자 로그인이 필요합니다.');
      CURRENT_ADMIN = admin;
      return JSON.stringify({ ok: true, data: ADMIN_API[name].apply(null, args) });
    }
    if (!PUBLIC_API[name]) throw new AppError('없는 기능입니다: ' + name);
    return JSON.stringify({ ok: true, data: PUBLIC_API[name].apply(null, args) });
  } catch (e) {
    if (!e.app) console.error(name, e && e.stack || e);
    return JSON.stringify({ ok: false, error: e.app ? e.message : '처리 중 오류가 발생했습니다. (' + (e && e.message) + ')', auth: /로그인이 필요/.test(e.message || '') });
  }
}
